🌪️ Weather Integration

# Merging 800+ flights with hourly wind data
weather_bank <- weather_list |>
  bind_rows() |>
  sort_values("weather_time")

final_df <- flights |>
  merge_asof(weather_bank,
    left_on = "departure_estimated",
    right_on = "weather_time",
    by = "departure_iata",
    direction = "nearest"
  )

Result: 97.9% weather match rate 🎯
